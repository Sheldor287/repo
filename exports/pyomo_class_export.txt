class Objectives:
    @staticmethod
    def max_flow(m):
        return pyo.Objective(expr=m.F, sense=pyo.maximize)

    @staticmethod
    def min_cost_flow(m):
        expr = sum(m.cost[u, v] * m.x[u, v] for (u, v) in m.edges)
        return pyo.Objective(expr=expr, sense=pyo.minimize)

    @classmethod
    def _apply(cls, problem_type, model):

        registry = {
            'max_flow': cls.max_flow,
            'min_cost_flow': cls.min_cost_flow,
        }
        if problem_type not in registry:
            raise ValueError(f"Unknown problem type: {problem_type}")
            
        model.obj = registry[problem_type](model)

class Rules:
    @staticmethod
    def edge_cap_rule(m, u, v):
        """Capacity can be bandwidth, time-requirement, or 1 if assessing count of independent paths"""
        return m.x[u, v] <= m.capacity[u, v]
    
    @staticmethod
    def edge_flow_rule(m, n):
        """Flow out = Flow in for all nodes; originator demand=-1, endpoint demand=1"""
        if n in (m.s, m.t):
            return pyo.Constraint.Skip
        inflow = sum(m.x[u, v] for (u, v) in m.edges if v == n)
        outflow = sum(m.x[u, v] for (u, v) in m.edges if u == n)
        return inflow == outflow
    
    @staticmethod
    def source_flow_rule(m):
        outflow = sum(m.x[u, v] for (u, v) in m.edges if u == m.s)
        inflow = sum(m.x[u, v] for (u, v) in m.edges if v == m.s)
        return outflow - inflow == m.F
    
    @staticmethod
    def sink_flow_rule(m):
        inflow = sum(m.x[u, v] for (u, v) in m.edges if v == m.t)
        outflow = sum(m.x[u, v] for (u, v) in m.edges if u == m.t)
        return inflow - outflow == m.F

    def get_residual_graph(self):
        R = nx.DiGraph()
        for u,v in self.model.edges:
            capacity = pyo.value(self.model.capacity[u,v])
            flow = pyo.value(self.model.x[u,v])
            forward = capacity - flow
            reverse = flow
            if forward > 0:
                R.add_edge(u,v,residual=forward)
            if reverse > 0:
                R.add_edge(v,u, residual=reverse)
        return R

        result = solver.solve(self.model)

    def _update_results(self, results):

        results = {
            'problem': {
                'lower_bound': results['Problem']()['Lower bound'],
                'upper_bound': results['Problem']()['Upper bound'],
                'obj_function_count': self.model.nobjectives(),
                'constraint_count': self.model.nconstraints(),
                'variable_count': self.model.nvariables(),
                'sense': str(self.model.obj.sense),
            },
            'solver': {
                'status': str(results['Solver']()['Status']),
                'termination_condition': str(
                    results['Solver']()['Termination condition']
                ),
                'termination_message': str(
                    results['Solver']()['Termination message']
                ),
            }
        }

        node_results = {}

        for n in self.g.nodes:

            demand = self.g.nodes[n].get('demand', 0)

            # Ignore zero-demand nodes
            if demand == 0:
                continue

            # Supply nodes
            if demand < 0:
                supply = -demand
                if (self.model.s, n) in self.model.edges:
                    satisfied = pyo.value(self.model.x[self.model.s, n])
                else:
                    satisfied = 0
                unused = supply - satisfied
                node_results[n] = {
                    'type': 'supply',
                    'demand': demand,
                    'supply': supply,
                    'satisfied': satisfied,
                    'unmet': 0,
                    'unused': unused,
                    'satisfaction_rate': satisfied / supply
                }

            # Demand nodes
            elif demand > 0:
                if (n, self.model.t) in self.model.edges:
                    satisfied = pyo.value(self.model.x[n, self.model.t])
                else:
                    satisfied = 0
                unmet = demand - satisfied
                node_results[n] = {
                    'type': 'demand',
                    'demand': demand,
                    'supply': 0,
                    'satisfied': satisfied,
                    'unmet': unmet,
                    'unused': 0,
                    'satisfaction_rate': satisfied / demand
                }

        total_supply = sum(r['supply'] for r in node_results.values() if r['type'] == 'supply')
        total_used_supply = sum(r['satisfied']for r in node_results.values() if r['type'] == 'supply')
        total_demand = sum(r['demand'] for r in node_results.values() if r['type'] == 'demand')
        total_satisfied = sum(r['satisfied'] for r in node_results.values() if r['type'] == 'demand')
        total_unmet = total_demand - total_satisfied

        results['metrics'] = {
            'total_supply': total_supply,
            'total_used_supply': total_used_supply,
            'total_unused_supply': total_supply - total_used_supply,
            'total_demand': total_demand,
            'total_satisfied': total_satisfied,
            'total_unmet': total_unmet,
            'total_satisfaction_rate': (total_satisfied / total_demand if total_demand > 0 else None),
            'node_results': node_results
        }

        return results


    def __configure_graph(self, source='s', sink='t', artifical_source='dS', artifical_sink='dT'):
        """
        Standardizes the graph for LPs including:
        1. Creating super-source/sink artifical nodes
        2. Standardizing cost/capacity parameters for edges
        3. Standardizing demand parameter for nodes 
        4. Create an artifical source/sink to preserve demand balance 
        """
        H = self.g.copy()
        if not H.is_directed():
            H = H.to_directed()

        for _,_,d in H.edges(data=True):
            d.setdefault('capacity', 1)
            d.setdefault('cost', 0)

        for n, data in H.nodes(data=True):
            data.setdefault('demand', 0)

        # Add super source/sink
        H.add_node(source, demand=0, layer=-1)
        H.add_node(sink, demand=0, layer=1+max(d['layer'] for _,d in H.nodes(data=True)))
        supply = [n for n in H.nodes if H.nodes[n].get('demand') < 0]
        demand = [n for n in H.nodes if H.nodes[n].get('demand') > 0]
        for n in supply:
            H.add_edge(source, n, capacity=-H.nodes[n]['demand'], cost=0, artificial=True)
        for n in demand:
            H.add_edge(n, sink, capacity=H.nodes[n]['demand'], cost=0, artificial=True)

        # Normalize demand balance
        demand_imbalance = sum(-H.nodes[n]['demand'] for n in supply) - sum(H.nodes[n]['demand'] for n in demand)
        if demand_imbalance > 0:
            H.add_node(artifical_sink, demand=0, artifical=True, layer=-1)
            H.add_edge(source, artifical_sink, capacity=demand_imbalance, cost=0, artifical=True)
            H.add_edge(artifical_sink, sink, capacity=demand_imbalance, cost=0, artifical=True)
        elif demand_imbalance < 0:
            H.add_node(artifical_source, demand=0, artifical=True, layer=1+max(d['layer'] for _,d in H.nodes(data=True)))
            H.add_edge(artifical_source, sink, capacity=-demand_imbalance, cost=0, artifical=True)
            H.add_edge(source, artifical_source, capacity=-demand_imbalance, cost=0, artifical=True)

        return H

class GraphDrawer:

    def __init__(self, lp):
        self.lp = lp
        self.g = lp.g
        self.model = lp.model
        self.bgar_cmap = mcolors.LinearSegmentedColormap.from_list("node_cmap", [(0.0, "#ff3333"), (1.0, "#0080ff")])

    def draw(self, mode="flow", pos=None, figsize=(24, 8), **kwargs):
        fig, ax = plt.subplots(figsize=figsize)
        if pos is None:
            pos = nx.multipartite_layout(self.g, subset_key="layer")

        node_data = self.lp.get_node_results()
        edge_data = self.lp.get_edge_results()

        cmap = plt.cm.gray_r
        cmap = mcolors.LinearSegmentedColormap.from_list("edge_cmap", cmap(np.linspace(0.15, 1.0, 256)))

        if mode == "flow":
            self._draw_flow(pos, node_data, edge_data, cmap, ax=ax, **kwargs)
        elif mode == "residual":
            self._draw_residual(pos, node_data, edge_data, ax=ax, cmap=cmap, **kwargs)
        elif mode == "demand":
            self._draw_demand(pos, node_data, edge_data, ax=ax, **kwargs)
        elif mode == "cost":
            self._draw_cost(pos, node_data, edge_data, cmap, ax=ax, **kwargs)
        else:
            raise ValueError(f"Unknown drawing mode: {mode}")

        ax.set_axis_off()
        plt.show()

    def _get_node_colors(self, node_data):
        node_colors = []
        for n in self.g.nodes:
            data = node_data.get(n, {})
            if "satisfaction_rate" in data:
                rate = max(0.0, min(1.0, data["satisfaction_rate"]))
                node_colors.append(self.bgar_cmap(rate))
            else:
                node_colors.append("#8f8f8f")#gray
        return node_colors

    def _add_node_colorbar(self, ax):
        """Adds a dedicated colorbar for Node Satisfaction on the left."""
        norm_node = mpl.colors.Normalize(vmin=0, vmax=1)
        sm_node = mpl.cm.ScalarMappable(norm=norm_node, cmap=self.bgar_cmap)
        sm_node.set_array([])
        cbar_node = ax.figure.colorbar(sm_node, ax=ax, location="left", pad=0.08, shrink=0.8)
        cbar_node.set_label("Node Satisfaction Rate")

    def _draw_flow(self, pos, node_data, edge_data, cmap, ax=None, **kwargs):
        if ax is None:
            fig, ax = plt.subplots()

        edge_utilization = []
        edge_widths = []
        for u, v in self.g.edges:
            data = edge_data[(u, v)]
            utilization = data["utilization"]
            edge_utilization.append(utilization)
            edge_widths.append(1 + 2 * utilization)

        node_colors = self._get_node_colors(node_data)

        nx.draw_networkx_nodes(
            self.g, pos, node_color=node_colors, node_size=500, ax=ax
        )
        nx.draw_networkx_edges(
            self.g,pos,width=edge_widths,arrows=True,edge_color=edge_utilization,edge_cmap=cmap,ax=ax,)
        nx.draw_networkx_labels(self.g, pos, ax=ax)

        # Edge Colorbar (Right)
        norm_edge = mpl.colors.Normalize(vmin=0, vmax=1)
        sm_edge = mpl.cm.ScalarMappable(norm=norm_edge, cmap=cmap)
        sm_edge.set_array([])

# Objectives & constraints
def mcf_obj(m):
    return sum(m.cost[u, v] * m.x[u, v] for (u, v) in m.edges)

def mf_obj(m):
    S = [n for n in m.nodes if m.demand[n] < 0]
    T = [n for n in m.nodes if m.demand[n] > 0]
    outflow = sum(sum(m.x[s, j] for (s2, j) in m.edges if s2 == s) for s in S)
    inflow = sum(sum(m.x[i, t] for (i, t2) in m.edges if t2 == t) for t in T)
    return outflow - inflow

def edge_cap_rule(m, u, v):
    return m.x[u, v] <= m.capacity[u, v]

def edge_flow_rule(m, n):
    if n in m.s or n in m.t:
        return pyo.Constraint.Skip
    inflow = sum(m.x[u, v] for (u, v) in m.edges if v == n)
    outflow = sum(m.x[u, v] for (u, v) in m.edges if u == n)
    return outflow - inflow == m.demand[n]

def source_flow_rule(m):
    outflow = sum(m.x[u, v] for (u, v) in m.edges if u in m.s)
    inflow = sum(m.x[u, v] for (u, v) in m.edges if v in m.s)
    return outflow - inflow == m.F

def sink_flow_rule(m):
    inflow = sum(m.x[u, v] for (u, v) in m.edges if v in m.t)
    outflow = sum(m.x[u, v] for (u, v) in m.edges if u in m.t)
    return inflow - outflow == m.F

def preprocess_demand_imbalance(G):
    total_supply = sum(G.nodes[n]['demand'] for n in G.nodes() if G.nodes[n]['demand'] < 0)
    total_demand = sum(G.nodes[n]['demand'] for n in G.nodes() if G.nodes[n]['demand'] > 0)
    imbalance = total_supply + total_demand
    max_layer = max(G.nodes[n]['layer'] for n in G.nodes())

    #  Add artificial nodes
    if imbalance > 0:
        # More demand
        G.add_node('dS', demand=-imbalance, layer=-1)
        for n in G.nodes():
            if G.nodes[n].get('layer') == max_layer:
                G.add_edge('dS', n, cost=0, capacity=imbalance)

    elif imbalance < 0:
        # More supply
        G.add_node('dT', demand=-imbalance, layer=-1)
        for n in G.nodes():
            if G.nodes[n].get('layer') == 0:
                G.add_edge(n, 'dT', cost=0, capacity=-imbalance)
    return G

def standard_min_cost_flow(G):
    """ Minimize cost of total travel over the graph """
    # Create artificial nodes if needed
    G = preprocess_demand_imbalance(G)

    # Sets & params
    model = pyo.ConcreteModel()
    model.nodes = pyo.Set(initialize=list(G.nodes()))
    model.edges = pyo.Set(initialize=list(G.edges()), dimen=2)
    model.cost = pyo.Param(model.edges,initialize={(u, v): G.edges[u, v]['cost'] for u, v in G.edges()})
    model.capacity = pyo.Param(model.edges,initialize={(u, v): G.edges[u, v]['capacity'] for u, v in G.edges()})
    model.demand = pyo.Param(model.nodes,initialize={n: G.nodes[n]['demand'] for n in G.nodes()})

    model.x = pyo.Var(model.edges, within=pyo.NonNegativeReals)

def standard_max_flow(G):
    """ Maximize total flow over the graph """
    # Create artificial nodes if needed
    #G = preprocess_demand_imbalance(G)

    # Sets & params
    model = pyo.ConcreteModel()
    model.nodes = pyo.Set(initialize=list(G.nodes()))
    model.edges = pyo.Set(initialize=list(G.edges()), dimen=2)
    model.capacity = pyo.Param(model.edges,initialize={(u, v): G.edges[u, v]['capacity'] for u, v in G.edges()})
    model.demand = pyo.Param(model.nodes,initialize={n: G.nodes[n]['demand'] for n in G.nodes()})
    model.s = pyo.Set(initialize=[n for n in model.nodes if model.demand[n] < 0])
    model.t = pyo.Set(initialize=[n for n in model.nodes if model.demand[n] > 0])

    # Vars
    model.x = pyo.Var(model.edges, within=pyo.NonNegativeReals)
    model.F = pyo.Var(within=pyo.NonNegativeReals)

    # Objective & constraints
    model.obj = pyo.Objective(expr=model.F, sense=pyo.maximize)
    model.cap = pyo.Constraint(model.edges, rule=edge_cap_rule)
    model.flow = pyo.Constraint(model.nodes, rule=edge_flow_rule)
    model.source = pyo.Constraint(rule=source_flow_rule)
    model.sink = pyo.Constraint(rule=sink_flow_rule)

    # Solve 
    solver = pyo.SolverFactory('appsi_highs')
    result = solver.solve(model)
    result['Problem']()['Number of constraints'] = model.nconstraints()
    result['Problem']()['Number of variables'] = model.nvariables()
    flow_dict = {(u, v): pyo.value(model.x[u, v]) for (u, v) in model.edges}
    return result, flow_dict

def create_layered_graph(layer_sizes=[1,2,2,1], fully_connect=True):
    """ Generate a graph from a list of layer sizes """
    G = nx.DiGraph()
    layers = []

    node_id = 0

    # Create nodes layer by layer
    for i, size in enumerate(layer_sizes):
        layer = []
        for j in range(size):
            G.add_node(node_id, layer=i)
            layer.append(node_id)
            node_id += 1
        layers.append(layer)

    # Add edges between layers
    for i in range(len(layers) - 1):
        for u in layers[i]:
            for v in layers[i + 1]:
                if fully_connect:
                    G.add_edge(u, v)

    return G, layers

def layered_layout(G, s, t):
    levels = {}
    visited = {s}
    queue = [(s, 0)]

    while queue:
        node, level = queue.pop(0)
        levels[node] = level
        for neighbor in G.successors(node):
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, level + 1))

    pos = {}
    layer_counts = {}
    for node, level in levels.items():
        layer_counts[level] = layer_counts.get(level, 0) + 1

    layer_positions = {}
    for level, count in layer_counts.items():
        xs = list(range(count))
        layer_positions[level] = xs

    for node, level in levels.items():
        idx = layer_positions[level].pop(0)
        pos[node] = (level, idx)

    return pos

def assign_properties(G, layer_sizes, constant_capacity=False, layer_role_assign=None, solid_layers=[2,3,4]):
    # Metaparameters
    largest_layer = max(layer_sizes)
    layers = [list(range(sum(layer_sizes[:i]), sum(layer_sizes[:i+1]))) for i in range(len(layer_sizes))]
    total_demand = np.random.randint(largest_layer, 3 * largest_layer)
    layer_0_nodes = [n for n in G.nodes() if G.nodes[n]['layer'] == 0]
    layer_n_nodes = [n for n in G.nodes() if G.nodes[n]['layer'] == len(layers) - 1]
    roles = ['A', 'B', 'C']
    power = ['low', 'medium', 'high']

    # Demand assignment
    for n in G.nodes():
        G.nodes[n]['demand'] = 0
    while -sum(G.nodes[n]['demand'] for n in layer_0_nodes) < total_demand:
        for n in layer_0_nodes:
            if -sum(G.nodes[n]['demand'] for n in layer_0_nodes) == total_demand:
                break
            if n in layers[0]:  # Source layer
                G.nodes[n]['demand'] -= np.random.randint(0, 2)
    while sum(G.nodes[n]['demand'] for n in layer_n_nodes) < total_demand:
        for n in layer_n_nodes:
            if sum(G.nodes[n]['demand'] for n in layer_n_nodes) == total_demand:
                break
            if n in layers[-1]:  # Sink layer
                G.nodes[n]['demand'] += np.random.randint(0, 2)

    # Node data
    for n in G.nodes():
        G.nodes[n]['probability1'] = np.random.uniform(0.6, 1)

            if not layer_role_assign:
                G.nodes[n]['role'] = np.random.choice(roles, size=np.random.randint(2, len(roles)+1), replace=False).tolist()
            else:
                G.nodes[n]['role'] = layer_role_assign[G.nodes[n]['layer']]
                G.nodes[n]['power'] = np.random.choice(power, size=2, replace=False).tolist()
            #print(f"node: {n}, power: {G.nodes[n]['power']}, roles: {G.nodes[n]['role']}")
    

    # Rewrite graph edges based on power and role attributes 
    G.clear_edges()
    for i in range(len(layers) - 1):
        for u in layers[i]:
            for v in layers[i + 1]:
                for role in roles:
                    for p in power:

                            continue
                        if (role in G.nodes[u]['role']) and (role in G.nodes[v]['role']) and (p in G.nodes[u]['power']) and (p in G.nodes[v]['power']):
                            if constant_capacity:

                            else:
                                G.add_edge(u, v, role=role, power=p, capacity=np.random.randint(1, 10), probability=np.random.uniform(0.6, 0.99))
                        


    # Edge data
    for u, v in G.edges():
        G.edges[u, v]['performance'] = np.random.rand()*100

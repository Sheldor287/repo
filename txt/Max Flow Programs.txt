def max_flow(G: nx.DiGraph, s, t):
    model = pyo.ConcreteModel()
    # Sets, Params, Vars
    model.N = pyo.Set(initialize=list(G.nodes))
    model.A = pyo.Set(initialize=list(G.edges), dimen=2)
    cap = {(i, j): G[i][j].get('capacity') for (i, j) in G.edges}
    model.cap = pyo.Param(model.A,initialize=cap,within=pyo.NonNegativeReals)
    model.x = pyo.Var(
        model.A, domain=pyo.NonNegativeReals,
        initialize=lambda m, i,j: random.uniform(0, m.cap[i,j]) 
    )

    # Objective: Maximize flow through the graph 
    def max_flow_obj(m):
        outflow = sum(m.x[s, j] for (s2, j) in m.A if s2 == s)
        inflow = sum(m.x[i, s] for (i, s2) in m.A if s2 == s)
        return outflow - inflow
    
    # Constraint: Capacity 
    def capacity_UB(m, i, j):
        return m.x[i, j] <= m.cap[i, j]

    # Constraint: Flow Balance
    def flow_balance(m, v):
        if v == s or v == t:
            return pyo.Constraint.Skip

        inflow = sum(m.x[i, v] for (i, v2) in m.A if v2 == v)
        outflow = sum(m.x[v, j] for (v2, j) in m.A if v2 == v)
        return inflow - outflow == 0
    
    model.obj = pyo.Objective(rule=max_flow_obj, sense=pyo.maximize)
    model.capacity = pyo.Constraint(model.A, rule=capacity_UB)
    model.flow_conservation = pyo.Constraint(model.N,rule=flow_balance)
    return model

import pyomo.environ as pyo
import time

def all_max_flow(model, solver_name='appsi_highs', max_iter=1000, tol=1e-6):
    alternative_supports = []  # store support patterns only
    objective_values = []
    total_start = time.time()

    # Sets of used edges to conserve memory
    always_used = None
    never_used = None

    for t in range(max_iter):
        if t % 100 == 0 and t > 0:
            time_100 = time.time()
            print(f"Iteration {t}, Total Time: {time_100 - total_start:.2f} seconds")
        results = pyo.SolverFactory(solver_name).solve(model)

        # Break --> All optimal solutions found
        if results.solver.termination_condition != pyo.TerminationCondition.optimal:
            break

        # Identify support edges
        support_edges = tuple(sorted(e for e in model.A if pyo.value(model.x[e]) > tol))
        if not support_edges:
            break

        # Store support instead of full flow
        alternative_supports.append(support_edges)
        objective_values.append(pyo.value(model.obj))

        # Update essential edges --> Essential for large graphs
        if t == 0:
            always_used = set(support_edges)
            never_used = set(model.A) - set(support_edges)
        else:
            always_used &= set(support_edges)
            never_used &= set(model.A) - set(support_edges)

        # Add constraint to exclude current support
        constraint_name = f'exclude_flow_{t}'
        model.add_component(
            constraint_name,
            pyo.Constraint(expr=sum(model.x[e] for e in support_edges) <= len(support_edges) - tol)
        )

    end_time = time.time()
    print(f"Total time for {t+1} iterations: {end_time - total_start:.2f} seconds")

    return alternative_supports, objective_values

def bfs(residual, s, t):
    parent = {s: None}
    q = deque([s])

    added_nodes = [] 
    
    while q:
        u = q.popleft()
        for v in residual[u]:
            if v not in parent and residual[u][v]['capacity'] > 0:
                parent[v] = u
                added_nodes.append(v)  
                if v == t:
                    return parent, added_nodes
                q.append(v)
    return None, added_nodes


def ford_fulkerson_steps(G, s, t):
    residual = nx.DiGraph()
    for u, v, data in G.edges(data=True):
        residual.add_edge(u, v, capacity=data['capacity'])
        residual.add_edge(v, u, capacity=0)

    steps = []
    flow = 0

    while True:
        parent, added_nodes = bfs(residual, s, t)
        if parent is None:
            break
        removed_nodes = list(parent.keys())

        path = []
        bottleneck = float('inf')
        v = t
        while v != s:
            u = parent[v]
            path.append((u, v))
            bottleneck = min(bottleneck, residual[u][v]['capacity'])
            v = u
        path.reverse()

        for u, v in path:
            residual[u][v]['capacity'] -= bottleneck
            residual[v][u]['capacity'] += bottleneck

        flow += bottleneck

        steps.append({
            'path': path,
            'flow': bottleneck,
            'residual': copy.deepcopy(residual),
            'total_flow': flow,
            'added_nodes': added_nodes,
            'removed_nodes': removed_nodes
        })

    return steps

def ensure_st_path_directed(G, s, t, cap_range=(5, 10)):
    if not nx.has_path(G, s, t):
        # Use BFS/DFS on undirected version to find a path
        UG = G.to_undirected()
        path = nx.shortest_path(UG, s, t)
        for u, v in zip(path[:-1], path[1:]):
            # Add edge in the "correct" direction if missing
            if not G.has_edge(u, v):
                G.add_edge(u, v, capacity=random.randint(*cap_range))
            # Also optionally add reverse edge to keep BFS exploring options
            if not G.has_edge(v, u):
                G.add_edge(v, u, capacity=0)
    return G

def layered_layout(G, s, t):
    levels = {}
    visited = {s}
    queue = [(s, 0)]

    while queue:
        node, level = queue.pop(0)
        levels[node] = level
        for neighbor in G.successors(node):
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, level + 1))

    pos = {}
    layer_counts = {}
    for node, level in levels.items():
        layer_counts[level] = layer_counts.get(level, 0) + 1

    layer_positions = {}
    for level, count in layer_counts.items():
        xs = list(range(count))
        layer_positions[level] = xs

    layer_indices = {}
    for node, level in levels.items():
        idx = layer_positions[level].pop(0)
        pos[node] = (level, idx)

    return pos

def generate_multipartite_graph(layer_sizes, edge_prob=0.5, directed=True):
    """
    Generate a k-partite graph.
    
    Parameters
    ----------
    layer_sizes : list[int]
        Number of nodes in each layer.
    edge_prob : float
        Probability of an edge between nodes in consecutive layers.
    directed : bool
        Whether the graph is directed (default True)
    
    Returns
    -------
    G : networkx.DiGraph or Graph
        Multipartite graph with 'subset' attribute per node.
    """
    G = nx.DiGraph() if directed else nx.Graph()
    
    node_counter = 0
    layers = []
    
    for subset, size in enumerate(layer_sizes):
        layer_nodes = list(range(node_counter, node_counter + size))
        for n in layer_nodes:
            G.add_node(n, subset=subset)
        layers.append(layer_nodes)
        node_counter += size
    
    # Connect nodes between consecutive layers
    for i in range(len(layers)-1):
        for u in layers[i]:
            for v in layers[i+1]:
                if random.random() < edge_prob:
                    G.add_edge(u, v)
    
    return G

from scipy.stats import gamma

def log_prior(k, lam):
    if k <= 0 or lam <= 0:
        return -np.inf
    return (
        gamma.logpdf(k, a=2, scale=1) +
        gamma.logpdf(lam, a=2, scale=1)
    )

def log_likelihood(k, lam, data):
    if k <= 0 or lam <= 0:
        return -np.inf
    
    return np.sum(
        np.log(k/lam) +
        (k-1)*np.log(data/lam) -
        (data/lam)**k
    )

def log_posterior(theta, data):
    k, lam = theta
    return log_likelihood(k, lam, data) + log_prior(k, lam)


def metropolis_2d(data, n_samples=10000, proposal_sd=0.1):
    
    samples = np.zeros((n_samples, 2))
    
    theta_current = np.array([2.0, 10.0])
    
    for i in range(n_samples):
        
        # multiplicative log-normal proposal
        proposal = theta_current * np.exp(
            np.random.normal(0, proposal_sd, size=2)
        )
        
        log_ratio = (
            log_posterior(proposal, data)
            - log_posterior(theta_current, data)
        )
        
        if np.log(np.random.rand()) < log_ratio:
            theta_current = proposal
            
        samples[i] = theta_current
    
    return samples

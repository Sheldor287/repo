from __future__ import annotations
from sqlalchemy.orm import Mapped, mapped_column, relationship
from sqlalchemy import String, Text
from app.database import Base
from uuid import uuid4

from typing import TYPE_CHECKING
if TYPE_CHECKING:
    from app.models.mbse.semantic import Semantic

class Element(Base):
    __tablename__ = "elements"
    __table_args__ = {"schema": "test"}

    # Canonical identifier for this element across all models.
    element_uuid: Mapped[str] = mapped_column(
        String(36),
        primary_key=True,
        default=lambda: str(uuid4()),
    )
    # Human-readable identifier or name.
    name: Mapped[str | None] = mapped_column(
        String(255),
        nullable=True,
    )
    # Definition or description of the element.
    definition: Mapped[str] = mapped_column(
        Text,
        nullable=False,
    )
    # Semantic relationships in which this element is the source.
    outgoing_semantics: Mapped[list["Semantic"]] = relationship(
        back_populates="source_element",
        foreign_keys="Semantic.source_element_uuid",
    )
    # Semantic relationships in which this element is the target.
    incoming_semantics: Mapped[list["Semantic"]] = relationship(
        back_populates="target_element",
        foreign_keys="Semantic.target_element_uuid",
    )
    def __repr__(self) -> str:
        return f"Element(element_uuid={self.element_uuid!r}, name={self.name!r})"

from __future__ import annotations
from sqlalchemy.orm import Mapped, mapped_column, relationship
from sqlalchemy import ForeignKey, String, UniqueConstraint
from app.database import Base
from uuid import uuid4

from typing import TYPE_CHECKING
if TYPE_CHECKING:
    from app.models.mbse.element import Element
    from app.models.mbse.relationship import Relationship

class Semantic(Base):
    __tablename__ = "semantic_relationships"
    __table_args__ = (
        UniqueConstraint(
            "source_element_uuid",
            "relationship_uuid",
            "target_element_uuid",
            name="uq_semantic_triple",
        ),
        {"schema": "test"},
    )
    # Identifier for this specific semantic triple.
    semantic_uuid: Mapped[str] = mapped_column(
        String(36),
        primary_key=True,
        default=lambda: str(uuid4()),
    )
    # Subject / source element.
    source_element_uuid: Mapped[str] = mapped_column(
        String(36),
        ForeignKey("test.elements.element_uuid"),
        nullable=False,
        index=True,
    )
    # Predicate / relationship type.
    relationship_uuid: Mapped[str] = mapped_column(
        String(36),
        ForeignKey("test.relationships.relationship_uuid"),
        nullable=False,
        index=True,
    )
    # Object / target element.
    target_element_uuid: Mapped[str] = mapped_column(
        String(36),
        ForeignKey("test.elements.element_uuid"),
        nullable=False,
        index=True,
    )
    # ORM navigation.
    source_element: Mapped["Element"] = relationship(
        back_populates="outgoing_semantics",
        foreign_keys=[source_element_uuid],
    )
    relationship_type: Mapped["Relationship"] = relationship(
        back_populates="semantics",
    )
    target_element: Mapped["Element"] = relationship(
        back_populates="incoming_semantics",
        foreign_keys=[target_element_uuid],
    )

    def __repr__(self) -> str:
        return (
            f"Semantic("
            f"semantic_uuid={self.semantic_uuid!r}, "
            f"source={self.source_element_uuid!r}, "
            f"relationship={self.relationship_uuid!r}, "
            f"target={self.target_element_uuid!r})"
        )

from __future__ import annotations
from sqlalchemy import create_engine, select, text
from sqlalchemy.orm import Session
import argparse, csv, os, sys
from pathlib import Path
from uuid import UUID

# Project imports
BACKEND_DIR = Path(__file__).resolve().parents[1]
if str(BACKEND_DIR) not in sys.path:
    sys.path.insert(0, str(BACKEND_DIR))
from app.database import Base
from app.models.mbse.element import Element
from app.models.mbse.relationship import Relationship
from app.models.mbse.semantic import Semantic

# Configuration
PROJECT_DIR = BACKEND_DIR.parent

DEFAULT_DATA_DIR = PROJECT_DIR / "tests" / "test_data"

EXPECTED_TABLES = {"test.elements", "test.relationships", "test.semantic_relationships",}

CSV_COLUMNS = {
    "relationships": ["relationship_uuid", "name", "definition",],
    "elements": ["element_uuid", "name", "definition",],
    "semantics": ["semantic_uuid", "source_element_uuid", "relationship_uuid", "target_element_uuid",],
}

# CSV loading
def read_csv(filepath: Path, expected_columns: list[str],) -> list[dict]:
    if not filepath.exists():
        raise FileNotFoundError(f"Required CSV file not found: {filepath}")
    with filepath.open("r", newline="", encoding="utf-8-sig",) as file:
        reader = csv.DictReader(file)
        if reader.fieldnames is None:
            raise ValueError(f"{filepath.name} is empty.")
        actual = set(reader.fieldnames)
        expected = set(expected_columns)
        if actual != expected:
            raise ValueError(
                f"Invalid columns in {filepath.name}.\nMissing: {expected - actual}\nUnexpected: {actual - expected}"
                )
        rows = []
        for line_number, row in enumerate(reader, start=2):
            if None in row:
                raise ValueError(
                    f"Unexpected extra columns in {filepath.name}, line {line_number}."
                )
            if any(value is None for value in row.values()):
                raise ValueError(
                    f"Missing field in {filepath.name}, line {line_number}."
                )
            rows.append(row)
    return rows

def load_csv_data(data_dir: Path) -> dict[str, list[dict]]:
    return {
        "relationships": read_csv(data_dir / "relationships.csv", CSV_COLUMNS["relationships"],),
        "elements": read_csv(data_dir / "elements.csv", CSV_COLUMNS["elements"],),
        "semantics": read_csv(data_dir / "semantic_relationships.csv", CSV_COLUMNS["semantics"],),
    }

# Data validation
def validate_uuid(value: str, field_name: str, row_number: int,) -> None:
    try:
        UUID(value)
    except (ValueError, TypeError, AttributeError) as exc:
        raise ValueError(f"Invalid UUID in {field_name}, row {row_number}: {value}") from exc

def validate_unique(rows: list[dict], field: str, dataset_name: str,) -> None:
    values = [row[field] for row in rows]
    if len(values) != len(set(values)):
        raise ValueError(f"Duplicate {field} values in {dataset_name}.")

def validate_data(data: dict[str, list[dict]]) -> None:
    relationships = data["relationships"]
    elements = data["elements"]
    semantics = data["semantics"]
    # Required fields
    for dataset_name, rows in data.items():
        for row_number, row in enumerate(rows, start=2):
            for field, value in row.items():
                if not value.strip():
                    raise ValueError(f"Empty value in {dataset_name}, field '{field}', row {row_number}.")
    # UUID validation
    uuid_fields = {
        "relationships": ["relationship_uuid"],
        "elements": ["element_uuid"],
        "semantics": ["semantic_uuid", "source_element_uuid", "relationship_uuid", "target_element_uuid",],
    }
    for dataset_name, fields in uuid_fields.items():
        for row_number, row in enumerate(data[dataset_name], start=2,):
            for field in fields:
                validate_uuid(row[field], field, row_number,)
    # Primary-key uniqueness
    validate_unique(relationships, "relationship_uuid", "relationships",)
    validate_unique(elements, "element_uuid", "elements",)
    validate_unique(semantics, "semantic_uuid", "semantics",)
    # Relationship triple uniqueness
    triples = [(row["source_element_uuid"], row["relationship_uuid"], row["target_element_uuid"],) for row in semantics]
    if len(triples) != len(set(triples)):
        raise ValueError("Duplicate semantic triples found.")
    # Referential integrity
    element_ids = {row["element_uuid"] for row in elements}
    relationship_ids = {row["relationship_uuid"] for row in relationships}
    for row_number, semantic in enumerate(semantics, start=2):
        if semantic["source_element_uuid"] not in element_ids:
            raise ValueError(f"Unknown source element in semantic row {row_number}.")
        if semantic["target_element_uuid"] not in element_ids:
            raise ValueError(f"Unknown target element in semantic row {row_number}.")
        if semantic["relationship_uuid"] not in relationship_ids:
            raise ValueError(f"Unknown relationship in semantic row {row_number}.")

# ORM conversion
def build_orm_objects(data: dict[str, list[dict]]) -> dict:
    relationship_objects = [Relationship(**row) for row in data["relationships"]]
    element_objects = [Element(**row) for row in data["elements"]]
    semantic_objects = [Semantic(**row) for row in data["semantics"]]
    return {"relationships": relationship_objects, "elements": element_objects, "semantics": semantic_objects,}

# Database operations
def check_metadata() -> None:
    missing = EXPECTED_TABLES - set(Base.metadata.tables)
    if missing:
        raise RuntimeError(
            "SQLAlchemy metadata is missing expected tables: " f"{missing}. Check that all MBSE models are imported."
            )

def initialize_database(database_url: str, objects: dict, upsert: bool = False, create_tables: bool = False,) -> None:
    engine = create_engine(database_url, pool_pre_ping=True,)
    try:
        # Initialize the test schema.
        with engine.begin() as connection:
            connection.execute(text("CREATE SCHEMA IF NOT EXISTS test")) 
        # Create tables if requested.
        if create_tables:
            Base.metadata.create_all(engine)
        # All inserts happen within one transaction.
        with Session(engine) as session:
            with session.begin():
                load_order = ["relationships", "elements", "semantics",]
                for dataset_name in load_order:
                    rows = objects[dataset_name]
                    if upsert:
                        #merge() inserts new primary keys and updates existing records with matching primary keys.
                        for obj in rows:
                            session.merge(obj)
                    else:
                        session.add_all(rows)
                    print(f"Staged {len(rows)} {dataset_name} records.")
        print("\nDatabase transaction committed.")
    finally:
        engine.dispose()

def main():
    parser = argparse.ArgumentParser(description="Load generated MBSE graph CSVs into PostgreSQL.")
    parser.add_argument(
        "--data-dir",
        type=Path,
        default=DEFAULT_DATA_DIR,
        help="Directory containing the three generated CSV files.",
    )
    parser.add_argument(
        "--dry-run",
        action="store_true",
        help="Validate CSV data without connecting to the database.",
    )
    parser.add_argument(
        "--upsert",
        action="store_true",
        help="Insert new records or update records with matching UUIDs.",
    )
    parser.add_argument(
        "--create-tables",
        action="store_true",
        help="Create missing tables using SQLAlchemy metadata.",
    )
    args = parser.parse_args()
    print(f"Reading CSV files from: {args.data_dir}")
    data = load_csv_data(args.data_dir)
    print("\nValidating data...")
    validate_data(data)
    print("Validation successful.")
    for name, rows in data.items():
        print(f"{name}: {len(rows)} records")
    if args.dry_run:
        print("\nDry run complete. No database changes made.")
        return
    database_url = os.getenv("PG_CONN")
    if not database_url:
        raise RuntimeError("PG_CONN environment variable is not set.")
    check_metadata()
    objects = build_orm_objects(data)
    initialize_database(
        database_url=database_url,
        objects=objects,
        upsert=args.upsert,
        create_tables=args.create_tables,
    )

if __name__ == "__main__":
    main()

from __future__ import annotations

import argparse
import csv
import random
from pathlib import Path
from uuid import UUID, uuid5
from faker import Faker

# Fixed namespace makes UUID5 identifiers reproducible across runs.
NAMESPACE = UUID("6ba7b810-9dad-11d1-80b4-00c04fd430c8")

RELATIONSHIPS = {
    "OwningMembership": ("A membership in which an owning element owns another element."),
    "FeatureMembership": ("A membership that relates an element to a feature."),
    "FeatureTyping": ("A relationship that specifies the type of a feature."),
    "Subsetting": ("A relationship in which one feature is a subset of another."),
    "Redefinition": ("A relationship in which one feature redefines another feature."),
}

# A small automotive vocabulary used to give the generated model some domain structure.
COMPONENT_TYPES = {
    "VehicleDefinition": ["Vehicle", "Automobile", "TransportPlatform",],
    "PowertrainDefinition": ["Powertrain", "PropulsionSystem", "DriveSystem",],
    "EngineDefinition": ["Engine", "ElectricMotor", "HybridMotor",],
    "TransmissionDefinition": ["Transmission", "Gearbox", "DriveUnit",],
    "EnergyStorageDefinition": ["Battery", "FuelTank", "EnergyStorageUnit",],
    "ControlSystemDefinition": ["Controller", "ControlModule", "ElectronicControlUnit",],
    "ChassisDefinition": ["Chassis", "Frame", "SuspensionAssembly",],
    "WheelDefinition": ["Wheel", "TireAssembly", "WheelAssembly",],
}

TYPE_DESCRIPTIONS = {
    "VehicleDefinition": "Defines a vehicle-level system.",
    "PowertrainDefinition": "Defines a system responsible for propulsion.",
    "EngineDefinition": "Defines a component that produces mechanical power.",
    "TransmissionDefinition": "Defines a component that transfers mechanical power.",
    "EnergyStorageDefinition": "Defines a component that stores usable energy.",
    "ControlSystemDefinition": "Defines a component responsible for system control.",
    "ChassisDefinition": "Defines the structural support system.",
    "WheelDefinition": "Defines a component used for vehicle movement.",
}

class TestModelGenerator:
    def __init__(self, n_elements: int = 100, seed: int = 123, output_dir: str | Path = "prod/data/initialization",):
        if n_elements < 1:
            raise ValueError("n_elements must be at least 1.")
        # Validate the number of elements.
        self.n_elements = n_elements
        self.seed = seed
        self.output_dir = Path(output_dir)
        # Initialize random number generators for reproducibility.
        self.rng = random.Random(seed)
        self.fake = Faker()
        self.fake.seed_instance(seed)
        # Initialize model data structures.
        self.elements: list[dict] = []
        self.semantics: list[dict] = []
        self.relationships: list[dict] = []
        # Initialize identifier tracking structures.
        self.element_ids: set[str] = set()
        self.semantic_triples: set[tuple] = set()
        # Initialize relationship identifier tracking.
        self.relationship_ids: dict[str, str] = {}

    # Identifier helpers
    @staticmethod
    def make_uuid(value: str) -> str:
        return str(uuid5(NAMESPACE, value))

    def element_uuid(self, name: str) -> str:
        return self.make_uuid(f"element:{name}")

    def relationship_uuid(self, name: str) -> str:
        return self.make_uuid(f"relationship:{name}")

    def semantic_uuid(self, source_id: str, relationship_id: str, target_id: str,) -> str:
        value = f"semantic:{source_id}:{relationship_id}:{target_id}"
        return self.make_uuid(value)

    # Element generation
    def add_element(self, name: str, definition: str,) -> str:
        element_id = self.element_uuid(name)
        if element_id in self.element_ids:
            return element_id
        self.element_ids.add(element_id)
        self.elements.append({"element_uuid": element_id, "name": name, "definition": definition,})
        return element_id

    def generate_type_elements(self) -> dict[str, str]:
        """Create the reusable type-definition elements."""
        type_ids = {}
        for type_name, description in TYPE_DESCRIPTIONS.items():
            type_ids[type_name] = self.add_element(name=type_name, definition=description,)
        return type_ids

    def generate_component_name(self, type_name: str, index: int,) -> str:
        """Generate a domain-specific component name."""
        options = COMPONENT_TYPES[type_name]
        base_name = self.rng.choice(options)
        # Include an index when a name has already been used.
        candidate = base_name
        if candidate in {element["name"] for element in self.elements}:
            candidate = f"{base_name}_{index}"
        return candidate

    def generate_component_definition(self, name: str, type_name: str,) -> str:
        """Generate a plausible component definition."""
        descriptions = {
            "VehicleDefinition": (f"{name} provides the overall vehicle platform. "),
            "PowertrainDefinition": (f"{name} provides mechanical power for vehicle movement. "),
            "EngineDefinition": (f"{name} converts stored energy into mechanical power. "),
            "TransmissionDefinition": (f"{name} transfers power to the drivetrain. "),
            "EnergyStorageDefinition": (f"{name} stores energy for vehicle operation. "),
            "ControlSystemDefinition": (f"{name} monitors and controls vehicle functions. "),
            "ChassisDefinition": (f"{name} provides structural support for vehicle components. "),
            "WheelDefinition": (f"{name} supports vehicle movement and road contact. "),}
        base = descriptions[type_name]
        additional = self.fake.sentence(nb_words=10, variable_nb_words=True,)
        return base + additional

    def generate_components(self, type_ids: dict[str, str],) -> list[dict]:
        """Generate component elements and their type assignments."""
        components = []
        type_names = list(COMPONENT_TYPES.keys())
        # The first element is always the vehicle root.
        for i in range(self.n_elements):
            if i == 0:
                type_name = "VehicleDefinition"
                name = "Vehicle"
            else:
                type_name = self.rng.choice(type_names[1:])
                name = self.generate_component_name(type_name, i)
            definition = self.generate_component_definition(name, type_name,)
            element_id = self.add_element(name, definition)
            component = {"element_uuid": element_id, "name": name, "type_name": type_name,}
            components.append(component)
            # Each component is typed by a type-definition element.
            self.add_semantic(source_id=element_id, relationship_name="FeatureTyping", target_id=type_ids[type_name],)
        return components

    # Relationship catalog
    def generate_relationship_catalog(self) -> None:
        """Generate the reusable relationship-type catalog."""
        for name, definition in RELATIONSHIPS.items():
            relationship_id = self.relationship_uuid(name)
            self.relationship_ids[name] = relationship_id
            self.relationships.append({
                "relationship_uuid": relationship_id,
                "name": name,
                "definition": definition,
            })

    # Semantic generation
    def add_semantic(self, source_id: str, relationship_name: str, target_id: str,) -> None:
        relationship_id = self.relationship_ids[relationship_name]
        triple = (source_id, relationship_id, target_id,)
        if triple in self.semantic_triples:
            return
        self.semantic_triples.add(triple)
        self.semantics.append({
            "semantic_uuid": self.semantic_uuid(*triple),
            "source_element_uuid": source_id,
            "relationship_uuid": relationship_id,
            "target_element_uuid": target_id,
        })

    def generate_ownership_relationships(self, components: list[dict],) -> None:
        """
        Construct a hierarchical ownership graph.
        Select a previously generated component as the parent.
        This guarantees that every component is reachable
        from the root through ownership relationships.
        """
        if len(components) < 2:
            return
        root_id = components[0]["element_uuid"]
        for i in range(1, len(components)):
            parent = self.rng.choice(components[:i])
            self.add_semantic(
                source_id=parent["element_uuid"],
                relationship_name="OwningMembership",
                target_id=components[i]["element_uuid"],
            )

    # Validation
    def validate(self) -> None:
        element_ids = {row["element_uuid"] for row in self.elements}
        relationship_ids = {row["relationship_uuid"] for row in self.relationships}
        for semantic in self.semantics:
            assert semantic["source_element_uuid"] in element_ids
            assert semantic["target_element_uuid"] in element_ids
            assert semantic["relationship_uuid"] in relationship_ids
        assert len(element_ids) == len(self.elements)
        assert len(relationship_ids) == len(self.relationships)
        assert len(self.semantic_triples) == len(self.semantics)

    # CSV export
    def write_csv(self, filename: str, rows: list[dict], columns: list[str],) -> None:
        filepath = self.output_dir / filename
        with filepath.open("w", newline="", encoding="utf-8",) as file:
            writer = csv.DictWriter(file, fieldnames=columns,)
            writer.writeheader()
            writer.writerows(rows)
        print(f"Created: {filepath} ({len(rows)} records)")

    def export(self) -> None:
        self.output_dir.mkdir(parents=True, exist_ok=True,)
        self.write_csv(
            "relationships.csv",
            self.relationships,
            ["relationship_uuid", "name", "definition",],
        )
        self.write_csv(
            "elements.csv",
            self.elements,
            ["element_uuid", "name", "definition",],
        )
        self.write_csv(
            "semantic_relationships.csv",
            self.semantics,
            ["semantic_uuid", "source_element_uuid", "relationship_uuid", "target_element_uuid",],
        )

    # Main execution
    def generate(self) -> None:
        print("Generating test model...")
        self.generate_relationship_catalog()
        type_ids = self.generate_type_elements()
        components = self.generate_components(type_ids)
        self.generate_ownership_relationships(components)
        self.validate()
        self.export()
        print("\nGeneration complete.")
        print(f"Component elements: {self.n_elements}")
        print(f"Total elements: {len(self.elements)}")
        print(f"Relationship types: {len(self.relationships)}")
        print(f"Semantic assertions: {len(self.semantics)}")

def main():
    parser = argparse.ArgumentParser(description="Generate synthetic MBSE graph CSV files.")
    parser.add_argument("--n-elements", type=int, default=100, help="Number of generated component elements.",)
    parser.add_argument("--seed", type=int, default=123, help="Random seed for reproducible generation.",)
    parser.add_argument("--output-dir", type=str, default="prod/tests/test_data", help="Directory for generated CSV files.",)
    args = parser.parse_args()
    generator = TestModelGenerator(n_elements=args.n_elements, seed=args.seed, output_dir=args.output_dir,)
    generator.generate()

if __name__ == "__main__":
    main()

"""
Text to Codebase Converter

Reconstruct a directory/file structure from a text file produced by
CodebaseToText.

Expected input format:

    Folder Structure
    --------------------------------------------------

    project/
        src/
            main.py

    File Contents
    --------------------------------------------------


    src/main.py
    File type: .py
    print("hello")


    --------------------------------------------------
    File End
    --------------------------------------------------

The reconstructed files are written relative to the output directory.
"""

import argparse
import os
import re
import sys


class TextToCodebase:
    """
    Reconstruct a codebase from a TXT file generated by CodebaseToText.
    """

    FILE_SEPARATOR = "-" * 50
    FILE_END_MARKER = f"{FILE_SEPARATOR}\nFile End\n{FILE_SEPARATOR}"

    def __init__(
        self,
        input_path,
        output_path,
        verbose=False,
        overwrite=False,
    ):
        """
        Args:
            input_path: Path to the CodebaseToText TXT file.
            output_path: Directory where the codebase will be reconstructed.
            verbose: Enable detailed logging.
            overwrite: Allow existing files to be overwritten.
        """
        self.input_path = os.path.abspath(input_path)
        self.output_path = os.path.abspath(output_path)
        self.verbose = verbose
        self.overwrite = overwrite

        self.files_created = 0
        self.files_skipped = 0
        self.directories_created = 0

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def reconstruct(self):
        """
        Read the input TXT file and reconstruct the codebase.

        Returns:
            Dictionary containing reconstruction statistics.
        """
        text = self._read_input()

        folder_structure, file_contents = self._split_sections(text)

        if self.verbose:
            print(f"Input file: {self.input_path}")
            print(f"Output directory: {self.output_path}")

        os.makedirs(self.output_path, exist_ok=True)

        # Recreate directories first.
        self._recreate_folder_structure(folder_structure)

        # Then recreate files with their contents.
        self._recreate_files(file_contents)

        if self.verbose:
            self._print_summary()

        return {
            "files_created": self.files_created,
            "files_skipped": self.files_skipped,
            "directories_created": self.directories_created,
        }

    # ------------------------------------------------------------------
    # Input handling
    # ------------------------------------------------------------------

    def _read_input(self):
        """Read the source TXT file."""
        try:
            with open(self.input_path, "r", encoding="utf-8") as file:
                return file.read()
        except OSError as e:
            raise OSError(
                f"Could not read input file '{self.input_path}': {e}"
            ) from e

    def _split_sections(self, text):
        """
        Split the input into Folder Structure and File Contents sections.
        """
        folder_header = "Folder Structure"
        contents_header = "File Contents"

        if folder_header not in text:
            raise ValueError(
                "Input file does not contain a 'Folder Structure' section."
            )

        if contents_header not in text:
            raise ValueError(
                "Input file does not contain a 'File Contents' section."
            )

        folder_start = text.index(folder_header) + len(folder_header)
        contents_start = text.index(contents_header)

        folder_structure = text[folder_start:contents_start]
        file_contents = text[
            contents_start + len(contents_header):
        ]

        return folder_structure, file_contents

    # ------------------------------------------------------------------
    # Folder reconstruction
    # ------------------------------------------------------------------

    def _recreate_folder_structure(self, folder_structure):
        """
        Parse the indented folder structure and create directories.

        The original converter uses four spaces per indentation level.
        """
        lines = folder_structure.splitlines()

        for line in lines:
            if not line.strip():
                continue

            # Ignore the section delimiter.
            if line.strip() == self.FILE_SEPARATOR:
                continue

            stripped = line.strip()

            # Only directory entries end with '/'.
            if not stripped.endswith("/"):
                continue

            indentation = len(line) - len(line.lstrip(" "))

            # Original converter uses 4 spaces per level.
            level = indentation // 4

            directory_name = stripped.rstrip("/")

            if not directory_name:
                continue

            # The first directory in the tree represents the root folder
            # of the original codebase. We do not need to recreate it as a
            # child of output_path unless it is actually represented by
            # file paths.
            #
            # Instead, directory paths are reconstructed relative to the
            # output root.
            relative_path = self._directory_from_tree(
                lines,
                line,
                level,
                directory_name,
            )

            if relative_path is None:
                continue

            self._create_directory(relative_path)

    def _directory_from_tree(
        self,
        lines,
        current_line,
        level,
        directory_name,
    ):
        """
        Determine the relative directory path represented by a tree entry.

        This method uses indentation to reconstruct parent directories.
        """
        current_index = lines.index(current_line)

        parents = []

        for previous_line in lines[:current_index]:
            if not previous_line.strip():
                continue

            if previous_line.strip() == self.FILE_SEPARATOR:
                continue

            previous_indent = len(previous_line) - len(
                previous_line.lstrip(" ")
            )

            previous_level = previous_indent // 4

            previous_name = previous_line.strip().rstrip("/")

            if not previous_name:
                continue

            if previous_level < level and previous_line.strip().endswith("/"):
                while len(parents) > previous_level:
                    parents.pop()

                parents.append(previous_name)

        parents = parents[:level]

        parts = parents + [directory_name]

        # Remove an accidental empty/root component.
        parts = [p for p in parts if p]

        if not parts:
            return None

        return os.path.join(*parts)

    def _create_directory(self, relative_path):
        """Create a directory relative to the output directory."""
        directory_path = os.path.join(
            self.output_path,
            relative_path,
        )

        if not os.path.exists(directory_path):
            os.makedirs(directory_path, exist_ok=True)
            self.directories_created += 1

            if self.verbose:
                print(f"Created directory: {directory_path}")

    # ------------------------------------------------------------------
    # File reconstruction
    # ------------------------------------------------------------------

    def _recreate_files(self, file_contents):
        """
        Parse all file blocks and recreate the corresponding files.
        """
        file_blocks = self._parse_file_blocks(file_contents)

        if self.verbose:
            print(f"Found {len(file_blocks)} file blocks.")

        for file_path, content in file_blocks:
            self._write_file(file_path, content)

    def _parse_file_blocks(self, file_contents):
        """
        Parse file blocks from the File Contents section.

        Each block has the form:

            relative/path.py
            File type: .py
            <contents>

            --------------------------------------------------
            File End
            --------------------------------------------------
        """
        blocks = []

        # Split on the exact File End delimiter.
        raw_blocks = file_contents.split(self.FILE_END_MARKER)

        for raw_block in raw_blocks:
            raw_block = raw_block.strip("\n\r ")

            if not raw_block:
                continue

            parsed = self._parse_single_file_block(raw_block)

            if parsed is not None:
                blocks.append(parsed)

        return blocks

    def _parse_single_file_block(self, block):
        """
        Parse one file block.

        Returns:
            (relative_path, content) or None
        """
        lines = block.splitlines()

        if not lines:
            return None

        # Locate the File type line.
        file_type_index = None

        for index, line in enumerate(lines):
            if line.startswith("File type:"):
                file_type_index = index
                break

        if file_type_index is None:
            if self.verbose:
                print("Warning: Could not find 'File type:' in block.")
            return None

        # Everything before File type is expected to be the path.
        path_lines = lines[:file_type_index]

        # The path is normally the first non-empty line.
        relative_path = None

        for line in path_lines:
            if line.strip():
                relative_path = line.strip()
                break

        if not relative_path:
            if self.verbose:
                print("Warning: Could not determine file path.")
            return None

        # Everything after File type is file content.
        content_lines = lines[file_type_index + 1:]

        content = "\n".join(content_lines)

        # Because the original formatter adds a newline immediately before
        # the separator, preserve the normal file-ending newline.
        content = content.rstrip("\r\n") + "\n"

        return relative_path, content

    def _write_file(self, relative_path, content):
        """
        Write a reconstructed file.
        """
        relative_path = self._sanitize_relative_path(relative_path)

        file_path = os.path.abspath(
            os.path.join(
                self.output_path,
                relative_path,
            )
        )

        # Protect against paths escaping the output directory.
        if not self._is_within_output_directory(file_path):
            raise ValueError(
                f"Unsafe file path detected: {relative_path}"
            )

        parent_directory = os.path.dirname(file_path)

        if parent_directory:
            os.makedirs(parent_directory, exist_ok=True)

        if os.path.exists(file_path) and not self.overwrite:
            self.files_skipped += 1

            if self.verbose:
                print(f"Skipping existing file: {file_path}")

            return

        try:
            with open(
                file_path,
                "w",
                encoding="utf-8",
                newline="",
            ) as file:
                file.write(content)

            self.files_created += 1

            if self.verbose:
                print(f"Created file: {file_path}")

        except OSError as e:
            raise OSError(
                f"Could not write '{file_path}': {e}"
            ) from e

    # ------------------------------------------------------------------
    # Path handling
    # ------------------------------------------------------------------

    def _sanitize_relative_path(self, path):
        """
        Normalize paths stored in the TXT file.

        The original converter uses os.path.relpath(), so paths may use
        either '/' or '\\' depending on the platform.
        """
        path = path.strip()

        # Normalize both Windows and POSIX separators.
        path = path.replace("\\", os.sep)
        path = path.replace("/", os.sep)

        # Remove accidental leading separators.
        path = path.lstrip("/\\")

        return os.path.normpath(path)

    def _is_within_output_directory(self, path):
        """
        Ensure a reconstructed path cannot escape output_path.
        """
        output = os.path.abspath(self.output_path)
        target = os.path.abspath(path)

        try:
            return os.path.commonpath([output, target]) == output
        except ValueError:
            # Different drives on Windows.
            return False

    # ------------------------------------------------------------------
    # Reporting
    # ------------------------------------------------------------------

    def _print_summary(self):
        """Print reconstruction statistics."""
        print()
        print("-" * 50)
        print("Reconstruction Complete")
        print("-" * 50)
        print(f"Output directory:   {self.output_path}")
        print(f"Directories created: {self.directories_created}")
        print(f"Files created:       {self.files_created}")
        print(f"Files skipped:       {self.files_skipped}")
        print("-" * 50)


def main():
    """CLI entry point."""
    parser = argparse.ArgumentParser(
        description=(
            "Reconstruct a codebase from a TXT file produced by "
            "CodebaseToText."
        ),
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:

  %(prog)s --input codebase.txt --output ./reconstructed

  %(prog)s --input codebase.txt --output ./project --overwrite

  %(prog)s --input codebase.txt --output ./project --verbose
        """,
    )

    parser.add_argument(
        "--input",
        required=True,
        help="TXT file produced by CodebaseToText.",
    )

    parser.add_argument(
        "--output",
        required=True,
        help="Directory where the codebase will be reconstructed.",
    )

    parser.add_argument(
        "--overwrite",
        action="store_true",
        help="Overwrite existing files.",
    )

    parser.add_argument(
        "--verbose",
        action="store_true",
        help="Show detailed processing information.",
    )

    args = parser.parse_args()

    try:
        text_to_codebase = TextToCodebase(
            input_path=args.input,
            output_path=args.output,
            verbose=args.verbose,
            overwrite=args.overwrite,
        )

        text_to_codebase.reconstruct()

        if args.verbose:
            print("Conversion completed successfully!")

        return 0

    except (OSError, ValueError) as e:
        print(f"Error: {e}")
        return 1


if __name__ == "__main__":
    sys.exit(main())
